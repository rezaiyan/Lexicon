6 different patterns → Unified base with event sink & OnEvents
Current
✗ 6 distinct ViewModel patterns across 13 ViewModels — no shared contract

✗ SubscriptionVM: 7 separate StateFlows combined late — hard to reason about atomicity
✗ No shared base class — every new ViewModel re-invents state management
✗ Side effects handled differently: Channel vs SharedFlow vs callback lambdas
≡ OnEvents composable already exists in :core — validates flow type at runtime
Proposed
✓ Base class: BaseViewModel<State, Effect> — state + effects, no sealed event dispatch
✓ One atomic mutableStateOf<State> per screen — Compose-native, no Flow-to-State bridge
✓ Flow<Effect> via Channel — one-shot effects collected by OnEvents
✓ @Composable fun state() — returns State<S> directly, Compose tracks reads automatically
✓ Event sink — ViewModel public methods passed as lambdas to composables
abstract class BaseViewModel<S, F> : ViewModel() {

    private val _state = mutableStateOf(initialState())

    @Composable
    fun state(): State<S> = _state // Compose snapshot — auto-recompose on change

    private val _effects = Channel<F>(BUFFERED)
    val effects: Flow<F> = _effects.receiveAsFlow()

    abstract fun initialState(): S

    protected fun updateState(reducer: S.() -> S) {
        _state.value = _state.value.reducer()
    }

    protected fun emitEffect(effect: F) {
        viewModelScope.launch { _effects.send(effect) }
    }

    // Try<T> integration — fold result directly into state
    protected fun <T> Try<T>.reduce(
        onSuccess: S.(T) -> S,
        onFailure: S.(Throwable) -> S
    ) {
        fold(
            onSuccess = { updateState { onSuccess(it) } },
            onFailure = { updateState { onFailure(it) } }
        )
    }
}

fun interface UseCase<in P, out R> {
    suspend operator fun invoke(params: P): Try<R>
}

// Flow use case — reactive streams
fun interface FlowUseCase<in P, out R> {
    operator fun invoke(params: P): Flow<R>
}

// Unit param alias
typealias NoParamUseCase<R> = UseCase<Unit, R>
typealias NoParamFlowUseCase<R> = FlowUseCase<Unit, R>

Repository Contract
Mixed error contracts → Consistent Try<T> / Flow<T>
Current
✗ IWordRepository mixes 3 error contracts in one interface
IWordRepository — 3 Contracts

suspend fun deleteWord(id: Int)               // ← throws!

// Contract 2: Try<T> — explicit error handling

✗ Callers of updateWord must wrap in Try { } themselves — inconsistent burden
✗ deleteWord(id) throws but deleteAllWords() returns Try — same operation, different contracts
Proposed
✓ Two rules — no exceptions
Rule 1: Suspend → always Try<T>
suspend fun updateWord(word: Word): Try<Word>
suspend fun deleteWord(id: Int): Try<Unit>
suspend fun insertWords(words: List<Word>): Try<Int>
suspend fun deleteAllWords(): Try<Unit>
suspend fun syncWithRemote(): Try<Unit>
Rule 2: Streaming → always Flow<T>
fun getAllWords(): Flow<List<Word>>
fun getDueCards(): Flow<List<Word>>
fun getProgressStats(): Flow<ProgressStats>

// Multi-step ops: sealed progress inside Flow
fun deleteWords(ids: List<Int>): Flow<DeleteWordsProgress>  // ✓ keep
✓ Callers never wonder "does this throw?" — the type tells them
✓ Try<T> forces callers to handle errors at every call site
≡ Reads that can't fail (getWordById) return T? directly — no wrapping needed

// Structured logger — replaces ad-hoc logNetwork()
interface AppLogger {
    fun d(tag: String, message: () -> String)
    fun e(tag: String, error: Throwable, message: () -> String)
    fun network(method: String, url: String, status: Int, durationMs: Long)
}

// Timing interceptor — automatic performance tracking
class TimingInterceptor(private val logger: AppLogger) {
    override fun intercept(chain: Chain): Response {
        val start = TimeSource.Monotonic.markNow()
        val response = chain.proceed(chain.request)
        logger.network(request.method, request.url, response.status, start.elapsedNow())
        return response
    }
}

// UiState<T> — reusable across all screens
sealed interface UiState<out T> {
    data object Loading : UiState<Nothing>
    data class Success<T>(val data: T) : UiState<T>
    data class Error(val message: String?) : UiState<Nothing>
    data object Empty : UiState<Nothing>
}
